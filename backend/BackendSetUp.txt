✅ 1. Create backend folder (if not already)

Inside your Xclone project:

mkdir backend
cd backend
✅ 2. Initialize Node project
npm init -y
✅ 3. Replace your package.json

Open package.json and paste your config (what you sent).
Make sure it looks like this (important parts only):

{
  "name": "backend",
  "type": "module",
  "main": "index.js",
  "scripts": {
    "dev": "node --watch src/server.js",
    "start": "node src/server.js"
  }
}
✅ 4. Install dependencies

Run this:

npm install @arcjet/inspect @arcjet/node @clerk/express cloudinary cors dotenv express express-async-handler mongoose multer
✅ 5. Create project structure

Now create a clean backend structure:

mkdir src
cd src
mkdir controllers routes models middleware config utils
touch server.js app.js

 const startServer = async () => {
  try {
    await connectDB();

    // 👉 INSERT DUMMY USER (only for testing)
    const existingUser = await User.findOne({ email: "test@example.com" });

    if (!existingUser) {
      const user = await User.create({
        clerkId: "test_clerk_" + Date.now(),
        email: "test@example.com",
        firstName: "Vikas",
        lastName: "Chauhan",
        username: "vikas_" + Date.now(),
        bio: "Dummy user for testing",
        location: "India",
      });

      console.log("Dummy user created ✅", user.username);
    } else {
      console.log("Dummy user already exists ⚠️");
    }

    // start server
    if (ENV.NODE_ENV !== "production") {
      app.listen(5001, () =>
        console.log("Server is up and running on PORT:", 5001)
      );
    }
  } catch (error) {
    console.error("Failed to start server:", error.message);
    process.exit(1);
  }
};
